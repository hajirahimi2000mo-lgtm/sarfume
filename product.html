(function () { "use strict"; // تابع فرمت قیمت به فارسی function formatPrice(p) { if (typeof p === "number") { return p.toLocaleString("fa-IR") + " تومان"; } return p || "تماس بگیرید"; } // دریافت آی‌دی محصول از Query Params آدرس بار function getProductIdFromURL() { const params = new URLSearchParams(window.location.search); return parseInt(params.get("id"), 10); } // دریافت دیتابیس محصولات function getProductList() { if (Array.isArray(window.products)) { return window.products; } return []; } // افزودن مستقیم به سبد خرید و ذخیره در LocalStorage function handleAddToCart(product) { let cart = []; try { cart = JSON.parse(localStorage.getItem("sarfume_cart")) || []; } catch (e) { cart = []; } const existingIndex = cart.findIndex((item) => item.id === product.id); if (existingIndex > -1) { cart[existingIndex].quantity = (cart[existingIndex].quantity || 1) + 1; } else { cart.push({ id: product.id, name: product.name, price: product.price, rawPrice: product.rawPrice || 0, image: product.image, quantity: 1, }); } localStorage.setItem("sarfume_cart", JSON.stringify(cart)); } // رندر قالب محصول function renderProductDetail() { const container = document.getElementById("productDetail"); if (!container) return; const productId = getProductIdFromURL(); const products = getProductList(); if (!productId || isNaN(productId)) { container.innerHTML = `

محصولی انتخاب نشده است!

لطفاً از طریق صفحه اصلی، عطر مورد نظر خود را انتخاب فرمایید.

بازگشت به ویترین ←
`; return; } const item = products.find((p) => p.id === productId); if (!item) { container.innerHTML = `

عطر مورد نظر یافت نشد!

ممکن است این رایحه به اتمام رسیده باشد یا آدرس اشتباه وارد شده باشد.

مشاهده سایر عطرها ←
`; return; } // به‌روزرسانی عنوان تب مرورگر document.title = `${item.name} | SARFUME`; // ساخت ساختار HTML کارت محصول container.innerHTML = `
${item.brand || "SARFUME EXCLUSIVE"}

${item.name}

${formatPrice(item.price || item.rawPrice)}
طبع و رایحه ${item.scent || "خاص و نیش"}
حجم شیشه ${item.volume || "۱۰۰ میل"}
دسته‌بندی ${item.category === "perfume" ? "ادکلن لوکس" : item.category === "spray" ? "اسپری بدن" : "بادی اسپلش"}

${item.description || "این ادکلن لوکس با خط بوی بی‌نظیر و ماندگاری بالا، نمادی از اصالت و جذابیت شما خواهد بود."}

`; // اتصال رویداد خرید const buyBtn = document.getElementById("addToCartBtn"); if (buyBtn) { buyBtn.addEventListener("click", () => { handleAddToCart(item); const originalText = buyBtn.textContent; buyBtn.textContent = "به سبد خرید افزوده شد ✓"; buyBtn.style.background = "#22c55e"; buyBtn.style.color = "#ffffff"; buyBtn.disabled = true; setTimeout(() => { buyBtn.textContent = originalText; buyBtn.style.background = "#d4af37"; buyBtn.style.color = "#0a0a0c"; buyBtn.disabled = false; }, 1500); }); } } // اجرای لودر در زمان مناسب if (document.readyState === "loading") { document.addEventListener("DOMContentLoaded", renderProductDetail); } else { renderProductDetail(); } })();
